local function mPlayer()
    sec("PLAYER")
    pcall(function()
        w("name: " .. LP.Name .. " | userid: " .. LP.UserId)
        local hrp = getHRP()
        if hrp then
            w("pos: " .. tostring(hrp.Position))

        if d then
            for _, v in ipairs(d:GetChildren()) do
                if v:IsA("Folder") then w("Data." .. v.Name .. "/ (" .. #v:GetChildren() .. " con)")

local function mRemotes()
    sec("REMOTES-RS")
    pcall(function()

        for _, v in ipairs(RS:GetDescendants()) do

local function mValues()

            or v:IsA("Vector3Value") or v:IsA("BrickColorValue") then

                if n > 2000 then w("-- ...cat o 2000") break end

local function mModules()
    if not CFG.megaModules then return end
    sec("MODULES (require thu)")
    pcall(function()
        local mods = {}
        for _, v in ipairs(RS:GetDescendants()) do
            if v:IsA("ModuleScript") then
                table.insert(mods, v)
                if #mods >= 120 then break end
            end
        end
        w("-- modules: " .. #mods)
        for _, m in ipairs(mods) do
            local ok, ret = pcall(require, m)
            if ok then
                if typeof(ret) == "table" then
                    local ks, n = {}, 0
                    for k in pairs(ret) do n = n + 1 if n <= 15 then table.insert(ks, tostring(k)) end end
                    w("OK  " .. m:GetFullName() .. "  table[" .. n .. "] keys: " .. table.concat(ks, ", "))
                else
                    w("OK  " .. m:GetFullName() .. "  -> " .. typeof(ret))
                end
            else
                w("ERR " .. m:GetFullName())
            end
        end
    end)
end
local function mSweep()
    if not CFG.megaSweep then return end
    sec("WORKSPACE SWEEP")
    pcall(function()
        local cR, cP, cC, cF = 0, 0, 0, 0
        for _, v in ipairs(WS:GetDescendants()) do
            if v:IsA("RemoteEvent") or v:IsA("RemoteFunction") then
                cR = cR + 1
                if cR <= 200 then w("REMOTE | " .. v:GetFullName()) end
            elseif v:IsA("ProximityPrompt") then
                cP = cP + 1
                if cP <= 500 then w("PROMPT | " .. v:GetFullName() .. "  [" .. tostring(v.ActionText) .. "]") end
            elseif v:IsA("ClickDetector") then
                cC = cC + 1
                if cC <= 500 then w("CLICK | " .. v:GetFullName()) end
            elseif (v:IsA("Model") or v:IsA("Tool")) and v.Name:lower():find("fruit", 1, true) then
                cF = cF + 1
                if cF <= 100 then w("FRUIT? | " .. v:GetFullName()) end
            end
        end
        w(string.format("-- remote:%d prompt:%d click:%d fruit:%d", cR, cP, cC, cF))
    end)
end

-- ---------- CHE DO 3+4: TOUR / RADAR ----------
local function runTour(radar)
    L = {}
    mInfo(radar and "RADAR" or "TOUR")
    mPlayer() mRemotes()
    sec("TOUR LOG")
    local seenE, seenN, seenI = {}, {}, {}
    local visited = {}
    local function isNew(p)
        for _, q in ipairs(visited) do
            local dx, dy, dz = p.X - q.X, p.Y - q.Y, p.Z - q.Z
            if math.sqrt(dx * dx + dy * dy + dz * dz) < 500 then return false end
        end
        return true
    end
    local hrp0 = getHRP()
    if not hrp0 then
        w("SKIP: khong thay nhan vat.")
    else
        local startCF = hrp0.CFrame
        local queue = {}
        for _, s in ipairs(STOPS) do
            table.insert(queue, { pos = Vector3.new(s[1], s[2], s[3]), tag = "quest" })
        end
        local stops, t0 = 0, os.clock()
        while #queue > 0 and stops < CFG.maxStops and os.clock() - t0 < 300 do
            if STOP then w("-- DUNG SOM theo yeu cau") break end
            stops = stops + 1
            local job = table.remove(queue, 1)
            local hrp = getHRP()
            if hrp then
                pcall(function()
                    hrp.CFrame = CFrame.new(job.pos + Vector3.new(0, 70, 0))
                    hrp.Anchored = true
                end)
            end
            task.wait(CFG.waitBase)
            local nE, nN, nI = 0, 0, 0
            pcall(function()
                local en = WS:FindFirstChild("Enemies")
                if en then for _, m in ipairs(en:GetChildren()) do
                    if not seenE[m.Name] then seenE[m.Name] = true; nE = nE + 1 end
                end end
                local np = WS:FindFirstChild("NPCs")
                if np then for _, x in ipairs(np:GetChildren()) do
                    if not seenN[x.Name] then
                        local ok, pv = pcall(function() return x:GetPivot().Position end)
                        seenN[x.Name] = ok and string.format("(%.0f,%.0f,%.0f)", pv.X, pv.Y, pv.Z) or "?"
                        nN = nN + 1
                    end
                end end
                if radar then
                    local map = WS:FindFirstChild("Map")
                    if map then for _, isl in ipairs(map:GetChildren()) do
                        if not seenI[isl.Name] then
                            local ok, pv = pcall(function()
                                if isl:IsA("BasePart") then return isl.Position end
                                return isl:GetPivot().Position
                            end)
                            if ok and pv and isNew(pv) then
                                seenI[isl.Name] = string.format("(%.0f,%.0f,%.0f)", pv.X, pv.Y, pv.Z)
                                table.insert(visited, pv)
                                table.insert(queue, { pos = pv, tag = "dao:" .. isl.Name })
                                nI = nI + 1
                            elseif ok then
                                seenI[isl.Name] = seenI[isl.Name] or "gan diem cu"
                            end
                        end
                    end end
                end
            end)
            if nE > 0 or nN > 0 or nI > 0 then task.wait(CFG.waitExtra) end
            table.insert(visited, job.pos)
            w(string.format("stop %d [%s] +quai:%d +npc:%d +dao:%d (doi:%d)", stops, job.tag, nE, nN, nI, #queue))
            status((radar and "RADAR " or "TOUR ") .. stops .. " diem (doi " .. #queue .. ")")
        end
        pcall(function()
            local hrp = getHRP()
            if hrp then hrp.CFrame = startCF; hrp.Anchored = false end
        end)
        w("-- xong " .. stops .. " diem")
    end
    sec("ISLANDS")
    local ni = {}
    for n in pairs(seenI) do table.insert(ni, n) end
    table.sort(ni)
    for _, n in ipairs(ni) do w(n .. " @ " .. seenI[n]) end
    sec("NPCS (ten + toa do)")
    local nn = {}
    for n in pairs(seenN) do table.insert(nn, n) end
    table.sort(nn)
    for _, n in ipairs(nn) do w(n .. " @ " .. seenN[n]) end
    sec("ENEMIES TOTAL")
    local ne = {}
    for n in pairs(seenE) do table.insert(ne, n) end
    table.sort(ne)
    for _, n in ipairs(ne) do w(n) end
    mMapMisc()
    saveFile(radar and "radar" or "tour")
end

-- ---------- CHE DO 5: SPY ----------
local function ser(v, depth, seen)
    depth = depth or 0
    seen = seen or {}
    local t = typeof(v)
    if t == "string" then return string.format("%q", v):sub(1, 120)
    elseif t == "number" or t == "boolean" or t == "nil" then return tostring(v)
    elseif t == "CFrame" then local p = v.Position return string.format("CF(%.1f,%.1f,%.1f)", p.X, p.Y, p.Z)
    elseif t == "Vector3" then return string.format("V3(%.1f,%.1f,%.1f)", v.X, v.Y, v.Z)
    elseif t == "Vector2" then return string.format("V2(%.1f,%.1f)", v.X, v.Y)
    elseif t == "Color3" then return string.format("C3(%d,%d,%d)", v.R * 255, v.G * 255, v.B * 255)
    elseif t == "EnumItem" then return tostring(v)
    elseif t == "Instance" then
        local ok, n = pcall(function() return v:GetFullName() end)
        return "INST:" .. (ok and n or v.ClassName)
    elseif t == "table" then
        if seen[v] or depth >= 3 then return "{...}" end
        seen[v] = true
        local parts, n = {}, 0
        for k, val in pairs(v) do
            n = n + 1
            if n > 8 then table.insert(parts, "...") break end
            table.insert(parts, "[" .. ser(k, depth + 1, seen) .. "]=" .. ser(val, depth + 1, seen))
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    elseif t == "function" then return "FUNC"
    else return t .. ":" .. tostring(v):sub(1, 60)
    end
end
local function runSpy()
    if not hookmetamethod then status("✘ Executor khong ho tro spy"); return end
    if _G.__SIKE_SPY_RUNNING then status("SPY dang chay roi"); return end
    _G.__SIKE_SPY_RUNNING = true
    local logs, counts, t0 = {}, {}, os.clock()
    local fname = string.format("SIKE_spy_%d_%s.txt", game.PlaceId, os.date("%Y%m%d_%H%M%S"))
    local done = false
    local function save()
        if done then return end
        done = true
        local out = { "SIKE SPY - " .. os.date() .. " - place " .. game.PlaceId, "tong: " .. #logs, "" }
        for _, l in ipairs(logs) do table.insert(out, l) end
        table.insert(out, "")
        table.insert(out, "===== THONG KE =====")
        local ks = {}
        for k in pairs(counts) do table.insert(ks, k) end
        table.sort(ks)
        for _, k in ipairs(ks) do table.insert(out, counts[k] .. "x  " .. k) end
        if writefile then
            pcall(writefile, fname, table.concat(out, "\n"))
            status("✔ SPY xong! File: " .. fname .. " (" .. #logs .. " dong)")
        else
            status("SPY xong (" .. #logs .. " dong) - khong ghi file duoc")
        end
        _G.__SIKE_SPY_RUNNING = false
    end
    status("SPY dang ghi " .. CFG.spyDur .. "s - lam gi trong game di bro!")
    local old
    old = hookmetamethod(game, "__namecall", function(self, ...)
        local method = getnamecallmethod()
        if (method == "InvokeServer" or method == "FireServer") and typeof(self) == "Instance" then
            local ok, isRS = pcall(function() return self:IsDescendantOf(RS) end)
            if ok and isRS and not done then
                local args = { ... }
                local parts = {}
                for i, a in ipairs(args) do
                    if i > 6 then table.insert(parts, "...") break end
                    table.insert(parts, ser(a))
                end
                local path = self:GetFullName()
                counts[path .. " | " .. method] = (counts[path .. " | " .. method] or 0) + 1
                table.insert(logs, string.format("[+%.1fs] %s | %s(%s)", os.clock() - t0, path, method, table.concat(parts, ", ")))
                if #logs >= CFG.spyMax then task.spawn(save) end
            end
        end
        return old(self, ...)
    end)
    task.delay(CFG.spyDur, save)
end

local function guard(job, name)
    if BUSY then status("Dang chay job khac - doi xong hoac bam STOP") return end
    BUSY = true
    STOP = false
    status(name .. " dang chay...")
    task.spawn(function()
        local ok, err = pcall(job)
        if not ok then status("✘ Loi: " .. tostring(err):sub(1, 100)) end
        BUSY = false
    end)
end
